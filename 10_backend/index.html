실행

terraform init
terraform apply
terraform output    # 11_infra_pipeline·12_app_pipeline이 remote_state로 읽는 값 확인

확인

aws s3 ls | grep aws-tf-cicd            # 버킷 2개
aws dynamodb list-tables                # lock 테이블

핵심 포인트

  • 이 단계의 terraform.tfstate는 로컬에 있습니다 — 이 단계 자체는 로컬 state.

파이프라인이 쓸 S3 backend는 이 단계가 "만드는 리소스"입니다.

  • outputs.tf의 값을 11_infra_pipeline·12_app_pipeline이 terraform_remote_state로 읽습니다.
  • tfstate 버킷은 prevent_destroy — 이후 단계 전부 지워도 여기는 보호됩니다.

# ============================================================
# 01-provider.tf — Terraform 코어·프로바이더 선언
# ============================================================

terraform {
  required_version = ">= 1.6.0, < 2.0.0"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.aws_region

  default_tags {
    tags = {
      Project   = "aws-tf"
      Stage     = "10_backend"
      ManagedBy = "terraform"
    }
  }
}

# 호출자 정보 — 계정 ID 참조용
data "aws_caller_identity" "current" {}

# ============================================================
# 02-state.tf — 모든 CI/CD의 기반이 되는 저장소 계층
#   tfstate 버킷 + DynamoDB lock + 파이프라인 artifact 버킷
#   → 11_infra_pipeline·12_app_pipeline이 outputs를 remote_state로 읽어간다
# ============================================================

# ---------- Terraform state backend (03_real이 사용) ----------
# S3 버킷 — state·artifact 저장소
resource "aws_s3_bucket" "tfstate" {
  bucket = "${var.project_name}-tfstate-${data.aws_caller_identity.current.account_id}"
  # destroy 금지 장치: state가 들어있는 버킷은 실수로 지우면 안 됨
  lifecycle {
    prevent_destroy = true
  }
}

# 버저닝 — state 실수 덮어쓰기 복구용
resource "aws_s3_bucket_versioning" "tfstate" {
  bucket = aws_s3_bucket.tfstate.id
  versioning_configuration { status = "Enabled" } # state 이력 복구용
}

# SSE — 저장 데이터 암호화
resource "aws_s3_bucket_server_side_encryption_configuration" "tfstate" {
  bucket = aws_s3_bucket.tfstate.id
  rule {
    apply_server_side_encryption_by_default { sse_algorithm = "AES256" }
  }
}

# 퍼블릭 차단 — state는 절대 공개 금지
resource "aws_s3_bucket_public_access_block" "tfstate" {
  bucket                  = aws_s3_bucket.tfstate.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

# DynamoDB lock — 동시 apply 방지 잠금 테이블
resource "aws_dynamodb_table" "tf_lock" {
  name         = "${var.project_name}-tf-lock"
  billing_mode = "PAY_PER_REQUEST" # state lock 전용 — 사용량 거의 없음
  hash_key     = "LockID"

  attribute {
    name = "LockID"
    type = "S"
  }
}

# ---------- 파이프라인 artifact 버킷 (11_infra_pipeline·12_app_pipeline 공용) ----------
# S3 버킷 — state·artifact 저장소
resource "aws_s3_bucket" "artifacts" {
  bucket        = "${var.project_name}-artifacts-${data.aws_caller_identity.current.account_id}"
  force_destroy = true # 실습용 — 비우고 삭제 가능하게
}

# 퍼블릭 차단 — state는 절대 공개 금지
resource "aws_s3_bucket_public_access_block" "artifacts" {
  bucket                  = aws_s3_bucket.artifacts.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

# ============================================================
# outputs.tf — 11_infra_pipeline·12_app_pipeline이 remote_state로 읽는 값
# ============================================================

output "artifacts_bucket" {
  description = "파이프라인 artifact 저장 버킷"
  value       = aws_s3_bucket.artifacts.bucket
}

output "artifacts_bucket_arn" {
  description = "파이프라인 IAM 정책에 쓸 artifact 버킷 ARN"
  value       = aws_s3_bucket.artifacts.arn
}

output "tfstate_bucket" {
  description = "Terraform state 저장 버킷 (backend 설정용)"
  value       = aws_s3_bucket.tfstate.bucket
}

output "tf_lock_table" {
  description = "state 잠금 DynamoDB 테이블"
  value       = aws_dynamodb_table.tf_lock.name
}

variable "aws_region" {
  description = "AWS 리전"
  type        = string
  default     = "ap-northeast-2"
}

variable "project_name" {
  description = "리소스 이름 접두사 (버킷·테이블 이름에 사용)"
  type        = string
  default     = "aws-tf-cicd"
}
